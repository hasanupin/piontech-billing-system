🧪 Testing Requirement — Feature Tests
// tests/Feature/RoleAccessTest.php
it('denies petugas from creating transfer transaction', function () {
    $petugas = User::factory()->create(['role' => Role::FieldOfficer]);
    $this->actingAs($petugas);
    expect($petugas->can('create', Transaction::class))->toBeTrue();
    // Tapi option transfer tidak tersedia — test via form options
});

it('denies petugas from editing pelanggan', function () {
    $petugas = User::factory()->create(['role' => Role::FieldOfficer]);
    $pelanggan = Customer::factory()->create();
    expect($petugas->can('update', $pelanggan))->toBeFalse();
});

it('only super_admin can manage users', function () {
    $admin = User::factory()->create(['role' => Role::Admin]);
    expect($admin->can('viewAny', User::class))->toBeFalse();
    $ceo = User::factory()->create(['role' => Role::SuperAdmin]);
    expect($ceo->can('viewAny', User::class))->toBeTrue();
});
