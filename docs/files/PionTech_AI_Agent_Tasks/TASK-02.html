📋 Migration: packages
Schema::create('packages', function (Blueprint $table) {
    $table->ulid('id')->primary();
    $table->string('name', 100);              // Label referensi: 'Paket 110'
    $table->decimal('default_price', 12, 2);  // Pre-fill saja — BUKAN constraint
    $table->tinyInteger('speed_mbps')->nullable();
    $table->text('description')->nullable();
    $table->boolean('is_active')->default(true);
    $table->timestamps();
});
ℹ️ Paket hanya referensi harga. Nominal transaksi diinput manual oleh admin (pre-fill dari default_price, tapi editable).
📋 Migration: clusters
Schema::create('clusters', function (Blueprint $table) {
    $table->ulid('id')->primary();
    $table->string('name', 100);              // 'Cluster A - Budi'
    $table->foreignId('officer_id')           // Petugas PIC
          ->constrained('users')->restrictOnDelete();
    $table->text('description')->nullable();
    $table->boolean('is_active')->default(true);
    $table->timestamps();
});
⚠️ Cluster adalah ACUAN penugasan pelanggan → petugas. BUKAN alamat. Ganti petugas = update officer_id di cluster, semua pelanggan ikut.
📋 Migration: customers
Schema::create('customers', function (Blueprint $table) {
    $table->ulid('id')->primary();
    $table->string('name', 150);
    $table->string('full_name')->nullable();
    $table->string('whatsapp_number', 20)->nullable();      // VARCHAR! Bukan INT (fix scientific notation Excel)
    $table->foreignUlid('cluster_id')
          ->constrained('clusters')->restrictOnDelete();
    $table->foreignUlid('package_id')->nullable()   // Nullable — referensi saja
          ->constrained('packages')->nullOnDelete();
    $table->string('address')->nullable();          // Referensi geografis (PADI, TALANG...)
    $table->text('maps_url')->nullable();
    $table->text('house_photo_url')->nullable();
    $table->tinyInteger('billing_day');        // 1-31, beda per pelanggan
    $table->enum('status', ['active','suspended','terminated'])->default('active');
    $table->date('suspended_at')->nullable();
    $table->date('registered_at');
    $table->text('notes')->nullable();
    $table->softDeletes();
    $table->timestamps();
    $table->index(['status','billing_day']);
});
📋 Migration: transactions
Schema::create('transactions', function (Blueprint $table) {
    $table->ulid('id')->primary();
    $table->foreignUlid('customer_id')->constrained('customers')->cascadeOnDelete();
    $table->foreignId('officer_id')->nullable()   // NULL saat payment_method=transfer
          ->constrained('users')->nullOnDelete();
    $table->date('period');                       // YYYY-MM-01
    $table->decimal('billed_amount', 12, 2);     // Input manual (pre-fill editable)
    $table->decimal('paid_amount', 12, 2);
    $table->enum('status', ['paid','partial','unpaid'])->default('paid');
    $table->enum('payment_method', ['cash','transfer'])->default('cash');
    $table->foreignId('recorded_by')->constrained('users');
    $table->dateTime('paid_at');
    $table->text('notes')->nullable();
    $table->timestamps();
    $table->unique(['customer_id','period']);    // 1 pelanggan 1 transaksi per bulan
    $table->index(['officer_id','period','payment_method']);
});
📋 Migration: officer_deposits
Schema::create('officer_deposits', function (Blueprint $table) {
    $table->ulid('id')->primary();
    $table->foreignId('officer_id')->constrained('users')->restrictOnDelete();
    $table->date('period');
    $table->decimal('amount', 12, 2);             // TITIP 1=1jt, TITIP 2=500rb...
    $table->foreignId('received_by')->constrained('users')->restrictOnDelete();
    $table->dateTime('deposited_at');
    $table->text('notes')->nullable();
    $table->timestamps();
    $table->index(['officer_id','period']);
});
