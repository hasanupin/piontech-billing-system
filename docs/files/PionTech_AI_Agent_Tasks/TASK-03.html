📋 Model: Customer — Global Scope Cluster
class Customer extends Model
{
    use SoftDeletes, HasUlids;

    protected static function booted(): void
    {
        static::addGlobalScope('cluster_scope', function (Builder $q) {
            $user = auth()->user();
            if ($user?->isRole(Role::FieldOfficer)) {
                $q->whereHas('clusters', fn($c) =>
                    $c->where('officer_id', $user->id));
            }
        });
    }

    public function scopeJatuhTempoHariIni(Builder $q): Builder
    {
        return $q->whereIn('status', ['active','suspended'])
                 ->where('billing_day', now()->day);
    }

    public function scopeBillable(Builder $q): Builder
    {
        // AKTIF + ISOLIR = semua yang masih ditagih. OFF tidak.
        return $q->whereIn('status', ['active','suspended']);
    }
}
📋 Model: Transaction — Auto Status & Transfer Rule
class Transaction extends Model
{
    use HasUlids;

    protected static function booted(): void
    {
        static::saving(function (self $t) {
            // Rule 1: auto status
            $t->status = match(true) {
                $t->paid_amount >= $t->billed_amount => 'paid',
                $t->paid_amount > 0                    => 'partial',
                default                                  => 'unpaid',
            };
            // Rule 2: transfer = tidak via petugas
            if ($t->payment_method === 'transfer') {
                $t->officer_id = null;
            }
        });
    }

    public function scopeTunai(Builder $q): Builder
    {
        return $q->where('payment_method', 'cash');
    }

    public function scopePeriode(Builder $q, $date): Builder
    {
        return $q->whereDate('period',
            Carbon::parse($date)->startOfMonth());
    }
}
📋 Service: BillingService
class BillingService
{
    public static function monthlySummary(Carbon $periode): array
    {
        $p = $periode->copy()->startOfMonth();
        $tunai    = Transaction::periode($p)->cash()->sum('paid_amount');
        $transfer = Transaction::periode($p)->where('payment_method','transfer')->sum('paid_amount');
        $setor    = OfficerDeposit::whereDate('period',$p)->sum('amount');

        return [
            'cash'           => $tunai,
            'transfer'        => $transfer,
            'total_terkumpul' => $tunai + $transfer,
            'sudah_disetor'   => $setor,
            'di_petugas'      => $tunai - $setor,   // FORMULA KUNCI
        ];
    }

    public static function officerRemainingBalance(int $petugasId, Carbon $periode): float
    {
        $p = $periode->copy()->startOfMonth();
        $tunai = Transaction::where('officer_id',$petugasId)
                 ->periode($p)->cash()->sum('paid_amount');
        $setor = OfficerDeposit::where('officer_id',$petugasId)
                 ->whereDate('period',$p)->sum('amount');
        return $tunai - $setor;
    }
}
