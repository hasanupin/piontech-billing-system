📋 Implementation
class CustomerImport implements ToModel, WithHeadingRow, WithValidation
{
    public function __construct(private string $clusterId) {}

    public function model(array $row): ?Pelanggan
    {
        return new Pelanggan([
            'name'       => trim($row['name']),
            'whatsapp_number'      => $this->cleanWhatsappNumber($row['wa'] ?? null),
            'cluster_id' => $this->clusterId,
            'package_id'   => $this->lookupPackage($row['paket'] ?? null),
            'address'     => $row['address'] ?? null,
            'maps_url'   => $this->validUrl($row['maps'] ?? null),
            'status'     => $this->cleanStatus($row['ket'] ?? 'active'),
            'billing_day' => 1, // default — admin edit manual nanti
            'registered_at'  => now()->toDateString(),
        ]);
    }

    private function cleanWhatsappNumber($v): ?string
    {
        if (!$v) return null;
        $s = is_float($v) ? sprintf('%.0f', $v) : (string)$v;
        $s = preg_replace('/\D/', '', $s);
        return str_starts_with($s, '62') ? $s
             : (str_starts_with($s,'0') ? '62'.substr($s,1) : $s);
    }
}
